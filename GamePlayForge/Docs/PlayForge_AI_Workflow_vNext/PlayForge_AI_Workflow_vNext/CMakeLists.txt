cmake_minimum_required(VERSION 3.20)
project(PlayForgeAIWorkflowReference LANGUAGES CXX)
add_executable(policy_gate_tests cpp/PolicyGateTests.cpp)
target_compile_features(policy_gate_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(policy_gate_tests PRIVATE /W4 /WX)
else()
  target_compile_options(policy_gate_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME policy_gate_reference COMMAND policy_gate_tests)
